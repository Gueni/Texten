
%---------------------------------------------------------------------------------------------------------------------------
%% Clear Console
%---------------------------------------------------------------------------------------------------------------------------
plecs('clc');                                           % Clear Console.
plecs('warning', 'warning message');                    % Display Warnings.

%---------------------------------------------------------------------------------------------------------------------------
%% Model variables from the model's own Initialization Commands
%---------------------------------------------------------------------------------------------------------------------------
% Runs the model's Initialization Commands in Octave's base workspace, collects every variable they define into the struct
% init_vars__, then removes them from the base workspace again.
% Why the base workspace : PLECS runs this script as a function, and sweep_callback below is a nested function of it. That
% makes the script's own workspace "static" in Octave 4.4 -- eval() cannot create variables in it, neither here nor inside
% a function (error: can not add variable "..." to a static workspace). The base workspace does not have that limit, and
% the model variables can never clash there with this script's own variables (data, scopes, mdl, ...).
init_cmds__     = plecs('get', plecs('get', '', 'CurrentCircuit'), 'InitializationCommands');
base_before__   = evalin('base', 'who');
init_vars__     = struct();
unwind_protect
    evalin('base', init_cmds__);
    init_names__ = setdiff(evalin('base', 'who'), base_before__);
    for k__ = 1:numel(init_names__)
        init_vars__.(init_names__{k__}) = evalin('base', init_names__{k__});
    end
unwind_protect_cleanup
    % Always leave the base workspace as it was, also when the Initialization Commands fail
    init_names__ = setdiff(evalin('base', 'who'), base_before__);
    if ~isempty(init_names__)
        evalin('base', ['clear ' strjoin(init_names__(:)', ' ')]);
    end
end_unwind_protect
clear init_cmds__ base_before__ init_names__ k__

%---------------------------------------------------------------------------------------------------------------------------
%% Configuration
%---------------------------------------------------------------------------------------------------------------------------
RESULTS_DIR     = '{{ scopes_folder }}';                % Directory for scopes traces
CSV_DIR         = '{{ csv_folder }}';              % Directory for csv time series results
HOLD_TRACES     = true;                                 % Set to true to hold traces after each simulation
CLEAR_TRACES    = true;                                 % Set to true to clear traces before starting
SAVE_TRACES     = true;                                 % Set to true to save traces to files
PARALLEL        = {{ 'true' if parallel else 'false' }};                                % true : all iterations run at once in parallel (PLECS 4.6+), false : one after another

%---------------------------------------------------------------------------------------------------------------------------
%% Parallel callback : PLECS calls this after each parallel simulation finishes
%---------------------------------------------------------------------------------------------------------------------------
% mdl, scopes and hold_traces are handed in through the anonymous function handle used in plecs('simulate', ...),
% because a function cannot see the script's own variables.
function status = sweep_callback(index, result, mdl, scopes, hold_traces)

    % Hold this iteration's traces under the same name the sequential loop uses
    if hold_traces
        trace_name = sprintf('Iter_%d_Standalone', index);
        for i = 1:length(scopes)
            try
                plecs('scope', [mdl '/' scopes{i}], 'HoldTrace', trace_name);
            catch
                % Silently ignore errors
            end
        end
    end

    % Keep only a small status instead of the full result (the data itself is already written by the ToFile block),
    % so a large sweep does not keep every simulation's waveforms in memory.
    if isstruct(result)
        status = true;
    else
        status = result;                                % PLECS returns the error message as a string if the run failed
    end
end

%---------------------------------------------------------------------------------------------------------------------------
%% Sweep data
%---------------------------------------------------------------------------------------------------------------------------
{% if mapvars_dim == 3 %}
data = {
    {% for row in mapvars %}
    { {% for inner in row %}[{{ inner | join(', ') }}]{% if not loop.last %}, {% endif %}{% endfor %} }{% if not loop.last %};
    {% endif %}
    {% endfor %}
};
{% else %}
data = [
    {% for row in mapvars %}    [{{ row | join(', ') }}];
    {% endfor %}
];
{% endif %}

%---------------------------------------------------------------------------------------------------------------------------
%% Simulation struct
%---------------------------------------------------------------------------------------------------------------------------
% Instead of copying every model variable into this script, they come from the model's Initialization Commands
% (init_vars__, collected at the top). Only the top-level variables the sweep changes are passed as ModelVars; all others
% keep the values the model's own Initialization Commands give them when the simulation starts.
modelvars_keep  = { {{ modelvars_keep | map('tojson') | join(', ') | replace('"', "'") }} };
simStruct       = struct();
simStruct.ModelVars = struct();
for k = 1:numel(modelvars_keep)
    if isfield(init_vars__, modelvars_keep{k})
        simStruct.ModelVars.(modelvars_keep{k}) = init_vars__.(modelvars_keep{k});
    end
end
simStruct.SolverOpts = {{ solveropts_struct }};

%---------------------------------------------------------------------------------------------------------------------------
%% Model
%---------------------------------------------------------------------------------------------------------------------------

mdl         = plecs('get', '', 'CurrentCircuit');               % Get model name and scope list
mdl_name    = (strsplit(mdl, filesep)){end};                    % Extract just the model name
mdl_name    = (strsplit(mdl_name, '.')){1};                     % Remove extension
scopes      = { {{ scopes | map('tojson') | join(', ') }} };    % List of scopes in the model%
t           = localtime(time());

%---------------------------------------------------------------------------------------------------------------------------
%% Clear Scopes
%---------------------------------------------------------------------------------------------------------------------------

% Clear all traces before starting if enabled
if CLEAR_TRACES
    for i = 1:length(scopes)
        try
            plecs('scope', [mdl '/' scopes{i}], 'ClearTraces');
        catch
            % Silently ignore errors
        end
    end
end

% Create results directory if saving is enabled
if SAVE_TRACES
    if ~exist(RESULTS_DIR, 'dir')
        mkdir(RESULTS_DIR);
        mkdir(CSV_DIR);
    end
end

%---------------------------------------------------------------------------------------------------------------------------
%% Header
%---------------------------------------------------------------------------------------------------------------------------
disp(sprintf('%s', '-' * ones(1, 90)));
disp(sprintf('%s', '-' * ones(1, 90)));
disp(sprintf('SIMULATION STARTED'));
disp(sprintf('%s', '-' * ones(1, 90)));
disp(sprintf('%s', '-' * ones(1, 90)));

disp(sprintf('Date               : %s', strftime("%r(%Z) %A %e %B %Y", t)));
disp(sprintf('Time               : %s', strftime("%H:%M:%S", t)));
disp(sprintf('Model              : %s', mdl_name));
disp(sprintf('Iterations         : %d', size(data, 1)));
if PARALLEL
    disp(sprintf('Execution          : parallel'));
else
    disp(sprintf('Execution          : sequential'));
end

%---------------------------------------------------------------------------------------------------------------------------
%% Main Loop
%---------------------------------------------------------------------------------------------------------------------------

% Parallel : the loop only prepares one simulation struct per iteration; they all run together after the loop.
% Every iteration starts again from the untouched baseStruct, so values set by one iteration never carry over to the next.
if PARALLEL
    simStructs  = cell(1, size(data, 1));
    baseStruct  = simStruct;
end

for sim = 1:size(data, 1)

    if PARALLEL
        simStruct = baseStruct;
    end

    %% ------------------------------------------------------------------------
    disp(sprintf('%s', '-' * ones(1, 90)));
    disp(sprintf('Iteration Number   : %d', sim));
    %% ------------------------------------------------------------------------

    {% for i, expr in mappings.items() %}
    % Assign value
    {{ expr }};

    % Print evaluated value
    disp(sprintf('%-50s = %12g','{{ expr.split("=")[0].strip() }}', {{ expr.split("=")[0].strip() }}));
    {% endfor %}

    % Construct filename using strcat with simulation number
    filename    = strcat('{{ FileNameStandalonePath }}', sprintf('%01d_Standalone.csv', sim));

    % Assign filename to ToFile block
    simStruct.ModelVars.Common.ToFile.FileNameStandalone = filename;

    % Print filename for verification
    disp(sprintf('Output File: %s', filename));

    % Parallel : store this iteration's struct (named like the held traces) and go on to the next one
    if PARALLEL
        simStruct.Name      = sprintf('Iter_%d_Standalone', sim);
        simStructs{sim}     = simStruct;
        continue;
    end

    %% ------------------------------------------------------------------------
    plecs('simulate', simStruct);
    %% ------------------------------------------------------------------------

    %% Hold traces for all scopes if enabled
    if HOLD_TRACES
        trace_name = sprintf('Iter_%d_Standalone', sim);
        for i = 1:length(scopes)
            try
                plecs('scope', [mdl '/' scopes{i}], 'HoldTrace', trace_name);
            catch
                % Silently ignore errors
            end
        end
    end
    %% ------------------------------------------------------------------------
end

%---------------------------------------------------------------------------------------------------------------------------
%% Parallel run : all prepared iterations at once
%---------------------------------------------------------------------------------------------------------------------------
if PARALLEL
    disp(sprintf('%s', '-' * ones(1, 90)));
    disp(sprintf('Running %d simulations in parallel ...', numel(simStructs)));

    out = plecs('simulate', simStructs, @(index, result) sweep_callback(index, result, mdl, scopes, HOLD_TRACES));

    % Report the iterations that failed (their cell holds the error message instead of true)
    n_failed = 0;
    for k = 1:numel(out)
        if ischar(out{k})
            n_failed = n_failed + 1;
            disp(sprintf('Iteration %d FAILED : %s', k, out{k}));
        end
    end
    disp(sprintf('Failed iterations  : %d of %d', n_failed, numel(out)));
end

%---------------------------------------------------------------------------------------------------------------------------
%% Save traces once per scope after all iterations
%---------------------------------------------------------------------------------------------------------------------------
if SAVE_TRACES
    for i = 1:length(scopes)
        try
            scope_path = [mdl '/' scopes{i}];
            % Replace any slashes in scope name for filename
            safe_scope_name = strrep(scopes{i}, '/', '_');
            safe_scope_name = [safe_scope_name '_Standalone'];
            trace_file = fullfile(RESULTS_DIR, sprintf('%s', safe_scope_name));
            plecs('scope', scope_path, 'SaveTraces', trace_file);
        catch
            % Silently ignore errors
        end
    end
end
%---------------------------------------------------------------------------------------------------------------------------
%% Footer
%---------------------------------------------------------------------------------------------------------------------------
disp(sprintf('%s', '-' * ones(1, 90)));
disp(sprintf('%s', '-' * ones(1, 90)));
disp(sprintf('SIMULATION FINISHED : Ran %d Iterations', size(data, 1)));
disp(sprintf('%s', '-' * ones(1, 90)));
disp(sprintf('%s', '-' * ones(1, 90)));
